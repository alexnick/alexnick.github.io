#!/usr/bin/env node

const fs = require("fs");
const path = require("path");

const color = {
    red: (s) => `\x1b[31m${s}\x1b[0m`,
    green: (s) => `\x1b[32m${s}\x1b[0m`,
    yellow: (s) => `\x1b[33m${s}\x1b[0m`,
    gray: (s) => `\x1b[90m${s}\x1b[0m`,
    bold: (s) => `\x1b[1m${s}\x1b[0m`
};

const SHORT_OBJECT_MAX_LEN = 140;

const ORDER = [
    "@P",
    "@ID",
    "id",

    "type",
    "mode",
    "action",
    "lookup",
    "xform.before",

    "group",
    "slice",
    "sound",
    "event_name",

    "size",
    "pos",
    "anchor",
    "pivot",
    "pivot.x",
    "pivot.y",
    "ratio",

    "layout",
    "props",
    "@O",
    "@H",

    "visible",
    "managed",
    "opacity",
    "background",
    "blend",

    "from",
    "from.x",
    "from.y",
    "to",
    "to.x",
    "to.y",

    "duration",
    "repeat",
    "interpolator",

    "states",
    "content"
];

let hasErrors = false;
let totalErrors = 0;
let checkedFiles = 0;

const inputArgs = process.argv.slice(2);

if (!inputArgs.length) {
    console.log(color.red("No files or directories provided."));
    console.log();
    console.log("Usage:");
    console.log("  node tools/jmm-check.js path/to/file.jmm");
    console.log("  node tools/jmm-check.js path/to/folder");
    console.log("  node tools/jmm-check.js .");
    process.exit(1);
}

const files = collectJmmFiles(inputArgs);

if (!files.length) {
    console.log(color.yellow("No .jmm files found."));
    process.exit(0);
}

for (const file of files) {
    checkedFiles += 1;

    const text = fs.readFileSync(file, "utf8");

    checkJson(file, text);
    checkTabs(file, text);
    checkTrailingWhitespace(file, text);
    checkSpacesAfterColonAndComma(file, text);
    checkMultilineSmallObjects(file, text);
    checkPropertyOrderInCompactObjects(file, text);
    checkContentLastInCompactObjects(file, text);
    checkHybridContentStyle(file, text);
}

console.log();
console.log(color.bold("JMM check summary"));
console.log(`${color.gray("Files checked:")} ${checkedFiles}`);
console.log(`${color.gray("Errors:")} ${totalErrors}`);

if (hasErrors) {
    console.log();
    console.log(color.red(color.bold("JMM check failed.")));
    process.exit(1);
}

console.log();
console.log(color.green(color.bold("JMM check passed.")));
process.exit(0);

function report(file, line, code, title, details = []) {
    hasErrors = true;
    totalErrors += 1;

    console.log();
    console.log(`${color.gray(file)}:${color.yellow(line)} ${color.red(color.bold(code))} ${color.bold(title)}`);

    for (const detail of details) {
        console.log(`  ${detail}`);
    }
}

function checkJson(file, text) {
    try {
        JSON.parse(text);
    } catch (e) {
        report(file, 1, "JMM000", "Invalid JSON/JMM syntax.", [
            `${color.gray("Reason:")} ${e.message}`,
            `${color.gray("Fix:")} check quotes, commas and brackets.`,
            `${color.gray("Note:")} pure JSON requires double quotes, not single quotes.`
        ]);
    }
}

function checkTabs(file, text) {
    lines(text).forEach((line, i) => {
        if (line.includes("\t")) {
            report(file, i + 1, "JMM001", "Tabs are not allowed.", [
                `${color.gray("Fix:")} replace tabs with spaces.`
            ]);
        }
    });
}

function checkTrailingWhitespace(file, text) {
    lines(text).forEach((line, i) => {
        if (/[ \t]+$/.test(line)) {
            report(file, i + 1, "JMM002", "Trailing whitespace.", [
                `${color.gray("Fix:")} remove spaces at the end of the line.`
            ]);
        }
    });
}

function checkSpacesAfterColonAndComma(file, text) {
    lines(text).forEach((line, i) => {
        const trimmed = line.trim();

        if (!trimmed || trimmed.startsWith("//")) return;

        const noStrings = stripStringsButKeepPunctuation(line);

        if (/"[^"]*"\s*:\S/.test(line) && !/"[^"]*"\s*:\s/.test(line)) {
            report(file, i + 1, "JMM003", "Missing space after colon.", [
                `${color.gray("Bad:")}  "type":"scale"`,
                `${color.gray("Good:")} "type": "scale"`
            ]);
        }

        if (/,[^\s}\]\r\n]/.test(noStrings)) {
            report(file, i + 1, "JMM004", "Missing space after comma.", [
                `${color.gray("Bad:")}  [1,2,3]`,
                `${color.gray("Good:")} [1, 2, 3]`
            ]);
        }
    });
}

function checkMultilineSmallObjects(file, text) {
    const ls = lines(text);

    for (let i = 0; i < ls.length; i++) {
        if (!looksLikeObjectStart(ls[i])) continue;

        const block = readObjectBlock(ls, i);
        if (!block) continue;

        const objectText = compactText(block.lines);
        const isMultiline = block.endLine > block.startLine;

        if (!isMultiline) continue;

        const hasContent = objectText.includes('"content"');
        const hasForceMultilineStructure =
            objectText.includes('"layout"') ||
            objectText.includes('"props"') ||
            objectText.includes('"states"') ||
            objectText.includes('"@O"') ||
            objectText.includes('"@H"');

        const isSmall = objectText.length <= SHORT_OBJECT_MAX_LEN;

        if (hasContent) continue;

        if (isSmall && !hasForceMultilineStructure) {
            report(file, i + 1, "JMM005", "Small object should probably be one line.", [
                `${color.gray("Length:")} ${objectText.length} chars`,
                `${color.gray("Rule:")} short simple objects should be compact.`,
                `${color.gray("Example:")} {"type": "delay", "duration": 300}`
            ]);
        }
    }
}

function checkPropertyOrderInCompactObjects(file, text) {
    lines(text).forEach((line, i) => {
        const keys = extractKeysFromCompactObject(line);

        if (keys.length < 2) return;

        const knownKeys = keys.filter((key) => getRank(key) !== 999);
        if (knownKeys.length < 2) return;

        const expectedKnownKeys = [...knownKeys].sort((a, b) => getRank(a) - getRank(b));

        for (let index = 0; index < knownKeys.length; index++) {
            if (knownKeys[index] !== expectedKnownKeys[index]) {
                const wrongKey = knownKeys[index];
                const expectedKey = expectedKnownKeys[index];

                report(file, i + 1, "JMM006", `Property "${wrongKey}" is out of order.`, [
                    `${color.gray("Current known order:")}  ${knownKeys.join(color.gray(" → "))}`,
                    `${color.gray("Expected known order:")} ${expectedKnownKeys.join(color.gray(" → "))}`,
                    `${color.gray("Fix:")} move ${color.yellow(`"${wrongKey}"`)} around ${color.yellow(`"${expectedKey}"`)}.`,
                    `${color.gray("Note:")} unknown fields are ignored and should not break the order check.`
                ]);

                return;
            }
        }
    });
}

function checkContentLastInCompactObjects(file, text) {
    lines(text).forEach((line, i) => {
        const keys = extractKeysFromCompactObject(line);

        if (!keys.includes("content")) return;

        if (keys[keys.length - 1] !== "content") {
            report(file, i + 1, "JMM007", `"content" should be the last property.`, [
                `${color.gray("Current:")} ${keys.join(color.gray(" → "))}`,
                `${color.gray("Rule:")} "content" usually contains children, so it should close the object.`
            ]);
        }
    });
}

function checkHybridContentStyle(file, text) {
    const ls = lines(text);

    for (let i = 0; i < ls.length; i++) {
        if (!looksLikeObjectStart(ls[i])) continue;

        const block = readObjectBlock(ls, i);
        if (!block) continue;

        const objectText = compactText(block.lines);

        if (!objectText.includes('"content"')) continue;

        const contentLineIndex = findLineIndex(block.lines, /"content"\s*:/);
        if (contentLineIndex === -1) continue;

        const absoluteContentLine = block.startLine + contentLineIndex;
        const contentLine = block.lines[contentLineIndex].trim();

        /*
            IMPORTANT FIX:

            Before, this checker used extractKeysFromText(block.lines.join("\n")).
            That collected keys recursively from children inside "content".

            Bad detected order:
                type -> repeat -> content -> type -> from -> to -> duration

            Correct shallow order for parent object:
                type -> repeat -> content

            So all order checks below now use only top-level keys of the current object.
        */
        const topLevelKeys = extractTopLevelKeysFromObjectText(block.lines.join("\n"));

        const contentIndex = topLevelKeys.indexOf("content");
        const keysBeforeContent = contentIndex === -1
            ? []
            : topLevelKeys.slice(0, contentIndex);

        if (contentIndex !== -1 && contentIndex !== topLevelKeys.length - 1) {
            report(file, absoluteContentLine + 1, "JMM008", `"content" should be the last property in content objects.`, [
                `${color.gray("Current detected order:")} ${topLevelKeys.join(color.gray(" → "))}`,
                `${color.gray("Rule:")} header fields first, "content": [ last.`
            ]);
        }

        if (!/"content"\s*:\s*\[\s*$/.test(contentLine)) {
            report(file, absoluteContentLine + 1, "JMM009", `"content" array should open on the same header line.`, [
                `${color.gray("Bad:")}`,
                `  "content":`,
                `  [`,
                `${color.gray("Good:")}`,
                `  "content": [`,
                `${color.gray("Rule:")} content uses hybrid style: compact object header, multiline content array.`
            ]);
        }

        const startsAsPlainMultilineObject = block.lines[0].trim() === "{";
        const hasForceMultilineStructure =
            objectText.includes('"layout"') ||
            objectText.includes('"props"') ||
            objectText.includes('"states"') ||
            objectText.includes('"@O"') ||
            objectText.includes('"@H"');

        if (startsAsPlainMultilineObject && !hasForceMultilineStructure && keysBeforeContent.length > 0) {
            report(file, block.startLine + 1, "JMM010", "Content object header should be compact.", [
                `${color.gray("Rule:")} objects with "content" usually keep fields before content on one header line.`,
                `${color.gray("Expected style:")} {"type": "...", "lookup": "...", "content": [`,
                `${color.gray("Then:")} content array items are formatted on following lines.`
            ]);
        }

        const knownKeysBeforeContent = keysBeforeContent.filter((key) => getRank(key) !== 999);
        const expectedKnownKeys = [...knownKeysBeforeContent].sort((a, b) => getRank(a) - getRank(b));

        for (let k = 0; k < knownKeysBeforeContent.length; k++) {
            if (knownKeysBeforeContent[k] !== expectedKnownKeys[k]) {
                report(file, absoluteContentLine + 1, "JMM011", `Header property "${knownKeysBeforeContent[k]}" is out of order before content.`, [
                    `${color.gray("Current known header order:")}  ${knownKeysBeforeContent.join(color.gray(" → "))}`,
                    `${color.gray("Expected known header order:")} ${expectedKnownKeys.join(color.gray(" → "))}`,
                    `${color.gray("Rule:")} content object header follows common JMM property order, then "content" comes last.`
                ]);
                break;
            }
        }
    }
}

function collectJmmFiles(args) {
    const result = [];

    for (const arg of args) {
        const normalized = path.resolve(process.cwd(), arg);

        if (!fs.existsSync(normalized)) {
            continue;
        }

        const stat = fs.statSync(normalized);

        if (stat.isFile()) {
            if (normalized.toLowerCase().endsWith(".jmm")) {
                result.push(normalized);
            }
            continue;
        }

        if (stat.isDirectory()) {
            walkDirectory(normalized, result);
        }
    }

    return unique(result);
}

function walkDirectory(dir, result) {
    const ignoredDirs = new Set([
        "node_modules",
        ".git",
        ".idea",
        "dist",
        "build",
        "coverage"
    ]);

    const entries = fs.readdirSync(dir, { withFileTypes: true });

    for (const entry of entries) {
        if (entry.isDirectory()) {
            if (ignoredDirs.has(entry.name)) continue;
            walkDirectory(path.join(dir, entry.name), result);
            continue;
        }

        if (entry.isFile() && entry.name.toLowerCase().endsWith(".jmm")) {
            result.push(path.join(dir, entry.name));
        }
    }
}

function unique(items) {
    return [...new Set(items)];
}

function looksLikeObjectStart(line) {
    const trimmed = line.trim();

    if (!trimmed) return false;

    return (
        trimmed === "{" ||
        trimmed.endsWith("{") ||
        /^[{]/.test(trimmed) ||
        /:\s*\{\s*$/.test(trimmed)
    );
}

function readObjectBlock(ls, startLine) {
    let depth = 0;
    const result = [];

    for (let i = startLine; i < ls.length; i++) {
        const line = ls[i];
        const noStrings = stripStrings(line);

        depth += countChar(noStrings, "{");
        depth -= countChar(noStrings, "}");

        result.push(line);

        if (depth === 0 && result.length > 0) {
            return {
                startLine,
                endLine: i,
                lines: result
            };
        }
    }

    return null;
}

function extractKeysFromCompactObject(line) {
    const trimmed = line.trim();

    if (!trimmed.includes("{") || !trimmed.includes("}")) return [];

    return extractTopLevelKeysFromObjectText(trimmed);
}

/*
    Extract keys only from the current object level.

    Example:

        {"type": "set", "repeat": 2, "content": [
            {"type": "opacity", "from": 1, "to": 0, "duration": 500}
        ]}

    Returns:

        ["type", "repeat", "content"]

    Not:

        ["type", "repeat", "content", "type", "from", "to", "duration"]
*/
function extractTopLevelKeysFromObjectText(text) {
    const firstBrace = text.indexOf("{");
    if (firstBrace === -1) return [];

    const keys = [];

    let objectDepth = 0;
    let arrayDepth = 0;

    for (let i = firstBrace; i < text.length; i++) {
        const ch = text[i];

        if (ch === '"' || ch === "'") {
            const stringStart = i;
            const stringEnd = findStringEnd(text, stringStart);

            if (stringEnd === -1) {
                break;
            }

            if (objectDepth === 1 && arrayDepth === 0) {
                const next = nextNonSpaceChar(text, stringEnd + 1);

                if (next && next.char === ":") {
                    keys.push(text.slice(stringStart + 1, stringEnd));
                }
            }

            i = stringEnd;
            continue;
        }

        if (ch === "{") {
            objectDepth += 1;
            continue;
        }

        if (ch === "}") {
            objectDepth -= 1;

            if (objectDepth === 0) {
                break;
            }

            continue;
        }

        if (ch === "[") {
            arrayDepth += 1;
            continue;
        }

        if (ch === "]") {
            arrayDepth -= 1;
            continue;
        }
    }

    return keys;
}

function findStringEnd(text, startIndex) {
    const quote = text[startIndex];
    let escaped = false;

    for (let i = startIndex + 1; i < text.length; i++) {
        const ch = text[i];

        if (escaped) {
            escaped = false;
            continue;
        }

        if (ch === "\\") {
            escaped = true;
            continue;
        }

        if (ch === quote) {
            return i;
        }
    }

    return -1;
}

function nextNonSpaceChar(text, startIndex) {
    for (let i = startIndex; i < text.length; i++) {
        if (!/\s/.test(text[i])) {
            return {
                char: text[i],
                index: i
            };
        }
    }

    return null;
}

function extractKeysFromText(text) {
    const keys = [];
    const regex = /"([^"]+)"\s*:/g;
    let match;

    while ((match = regex.exec(text))) {
        keys.push(match[1]);
    }

    return keys;
}

function extractKeysBeforeContent(blockLines) {
    const joined = blockLines.join("\n");
    const contentIndex = joined.search(/"content"\s*:/);

    if (contentIndex === -1) return [];

    const beforeContent = joined.slice(0, contentIndex);

    return extractKeysFromText(beforeContent);
}

function findLineIndex(ls, regex) {
    for (let i = 0; i < ls.length; i++) {
        if (regex.test(ls[i])) return i;
    }

    return -1;
}

function getRank(key) {
    const index = ORDER.indexOf(key);
    return index === -1 ? 999 : index;
}

function lines(text) {
    return text.split(/\r?\n/);
}

function compactText(ls) {
    return ls.map((line) => line.trim()).join(" ").replace(/\s+/g, " ").trim();
}

function countChar(text, char) {
    return [...text].filter((c) => c === char).length;
}

function stripStrings(text) {
    return text.replace(/"(?:\\.|[^"\\])*"/g, '""');
}

function stripStringsButKeepPunctuation(text) {
    return text.replace(/"(?:\\.|[^"\\])*"/g, '""');
}
