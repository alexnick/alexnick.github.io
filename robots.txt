#!/usr/bin/env node

const fs = require("fs");
const chalk = require("chalk");

const files = process.argv.slice(2);
let hasErrors = false;

const SHORT_OBJECT_MAX_LEN = 140;

const ORDER = [
    "@P", "@ID", "id",

    "type", "mode", "action", "lookup",
    "xform.before",

    "group", "slice", "sound", "event_name",

    "size", "pos", "anchor",
    "pivot", "pivot.x", "pivot.y",
    "ratio",

    "layout", "props", "@O", "@H",

    "visible", "managed", "opacity", "background", "blend",

    "from", "from.x", "from.y",
    "to", "to.x", "to.y",

    "duration", "repeat", "interpolator",

    "states",
    "content"
];

if (!files.length) {
    console.log(chalk.red("No files provided."));
    console.log();
    console.log("Usage:");
    console.log("  node tools/jmm-check.js path/to/file.jmm");
    console.log("  node tools/jmm-check.js src/**/*.jmm");
    process.exit(1);
}

for (const file of files) {
    const text = fs.readFileSync(file, "utf8");

    checkJson(file, text);
    checkTabs(file, text);
    checkTrailingWhitespace(file, text);
    checkSpacesAfterColonAndComma(file, text);
    checkMultilineSmallObjects(file, text);
    checkPropertyOrder(file, text);
    checkContentLast(file, text);
}

if (hasErrors) {
    console.log();
    console.log(chalk.red.bold("JMM check failed."));
    process.exit(1);
}

console.log(chalk.green.bold("JMM check passed."));
process.exit(0);

function report(file, line, code, title, details = []) {
    hasErrors = true;

    console.log();
    console.log(`${chalk.gray(file)}:${chalk.yellow(line)} ${chalk.red.bold(code)} ${chalk.bold(title)}`);

    for (const detail of details) {
        console.log(`  ${detail}`);
    }
}

function checkJson(file, text) {
    try {
        JSON.parse(text);
    } catch (e) {
        report(file, 1, "JMM000", "Invalid JSON/JMM syntax.", [
            `${chalk.gray("Reason:")} ${e.message}`,
            `${chalk.gray("Fix:")} check quotes, commas and brackets. Remember: JSON requires double quotes.`
        ]);
    }
}

function checkTabs(file, text) {
    lines(text).forEach((line, i) => {
        if (line.includes("\t")) {
            report(file, i + 1, "JMM001", "Tabs are not allowed.", [
                `${chalk.gray("Fix:")} replace tabs with spaces.`
            ]);
        }
    });
}

function checkTrailingWhitespace(file, text) {
    lines(text).forEach((line, i) => {
        if (/[ \t]+$/.test(line)) {
            report(file, i + 1, "JMM002", "Trailing whitespace.", [
                `${chalk.gray("Fix:")} remove spaces at the end of the line.`
            ]);
        }
    });
}

function checkSpacesAfterColonAndComma(file, text) {
    lines(text).forEach((line, i) => {
        const trimmed = line.trim();

        if (!trimmed || trimmed.startsWith("//")) return;

        if (/"[^"]+"\s*:[^\s{}\[\]",0-9\-truefalsenull]/.test(line)) {
            report(file, i + 1, "JMM003", "Missing space after colon.", [
                `${chalk.gray("Bad:")}  "type":"scale"`,
                `${chalk.gray("Good:")} "type": "scale"`
            ]);
        }

        if (/,[^\s{}\[\]"]/g.test(line)) {
            report(file, i + 1, "JMM004", "Missing space after comma.", [
                `${chalk.gray("Bad:")}  [1,2,3]`,
                `${chalk.gray("Good:")} [1, 2, 3]`
            ]);
        }
    });
}

function checkMultilineSmallObjects(file, text) {
    const ls = lines(text);

    for (let i = 0; i < ls.length; i++) {
        if (!ls[i].trim().endsWith("{")) continue;

        const startLine = i;
        let depth = 0;
        let collected = "";

        for (let j = i; j < ls.length; j++) {
            const noStrings = stripStrings(ls[j]);

            depth += countChar(noStrings, "{");
            depth -= countChar(noStrings, "}");

            collected += ls[j].trim() + " ";

            if (depth === 0) {
                const objectText = collected.trim();
                const isMultiline = j > startLine;

                const shouldStayMultiline =
                    objectText.includes('"content"') ||
                    objectText.includes('"layout"') ||
                    objectText.includes('"props"') ||
                    objectText.includes('"states"') ||
                    objectText.includes('"@O"') ||
                    objectText.includes('"@H"');

                const isSmall = objectText.length <= SHORT_OBJECT_MAX_LEN;

                if (isMultiline && isSmall && !shouldStayMultiline) {
                    report(file, startLine + 1, "JMM005", "Small object should probably be one line.", [
                        `${chalk.gray("Length:")} ${objectText.length} chars`,
                        `${chalk.gray("Rule:")} short simple objects should be compact.`,
                        `${chalk.gray("Example:")} {"type": "delay", "duration": 300}`
                    ]);
                }

                break;
            }
        }
    }
}

function checkPropertyOrder(file, text) {
    lines(text).forEach((line, i) => {
        const keys = extractKeysFromCompactObject(line);

        if (keys.length < 2) return;

        const knownKeys = keys.filter((key) => getRank(key) !== 999);
        if (knownKeys.length < 2) return;

        const expectedKnownKeys = [...knownKeys].sort((a, b) => getRank(a) - getRank(b));

        for (let index = 0; index < knownKeys.length; index++) {
            if (knownKeys[index] !== expectedKnownKeys[index]) {
                const wrongKey = knownKeys[index];
                const expectedKey = expectedKnownKeys[index];

                report(file, i + 1, "JMM006", `Property "${wrongKey}" is out of order.`, [
                    `${chalk.gray("Current known order:")}  ${knownKeys.join(chalk.gray(" → "))}`,
                    `${chalk.gray("Expected known order:")} ${expectedKnownKeys.join(chalk.gray(" → "))}`,
                    `${chalk.gray("Fix:")} move ${chalk.yellow(`"${wrongKey}"`)} around ${chalk.yellow(`"${expectedKey}"`)}.`,
                    `${chalk.gray("Note:")} unknown fields are ignored and should not break the order check.`
                ]);

                return;
            }
        }
    });
}

function checkContentLast(file, text) {
    lines(text).forEach((line, i) => {
        const keys = extractKeysFromCompactObject(line);

        if (!keys.includes("content")) return;

        if (keys[keys.length - 1] !== "content") {
            report(file, i + 1, "JMM007", `"content" should be the last property.`, [
                `${chalk.gray("Current:")} ${keys.join(chalk.gray(" → "))}`,
                `${chalk.gray("Rule:")} "content" usually contains children, so it should close the object.`
            ]);
        }
    });
}

function extractKeysFromCompactObject(line) {
    const trimmed = line.trim();

    if (!trimmed.includes("{") || !trimmed.includes("}")) return [];

    const keys = [];
    const regex = /"([^"]+)"\s*:/g;
    let match;

    while ((match = regex.exec(trimmed))) {
        keys.push(match[1]);
    }

    return keys;
}

function getRank(key) {
    const index = ORDER.indexOf(key);
    return index === -1 ? 999 : index;
}

function lines(text) {
    return text.split(/\r?\n/);
}

function countChar(text, char) {
    return [...text].filter((c) => c === char).length;
}

function stripStrings(text) {
    return text.replace(/"(?:\\.|[^"\\])*"/g, '""');
}
